/// طبقة التخزين: قاعدة بيانات Hive مشفرة.
/// مفتاح التشفير يُحفظ في مخزن الجهاز الآمن ولا يخرج منه.
class Store {
  Store._();

  static const _keyAlias = 'unimate_db_key';
  static late Box<Map> _box;

  static Future<void> init() async {
    await Hive.initFlutter();
    const secure = FlutterSecureStorage();
    var encoded = await secure.read(key: _keyAlias);
    if (encoded == null) {
      final rnd = Random.secure();
      final bytes = List<int>.generate(32, (_) => rnd.nextInt(256));
      encoded = base64Encode(bytes);
      await secure.write(key: _keyAlias, value: encoded);
    }
    _box = await Hive.openBox<Map>(
      'unimate_db',
      encryptionCipher: HiveAesCipher(base64Decode(encoded)),
    );
  }

  static ValueListenable<Box<Map>> get listenable => _box.listenable();

  static String newId() =>
      DateTime.now().microsecondsSinceEpoch.toRadixString(36) +
      Random().nextInt(9999).toString();

  // ---------- الملف الشخصي ----------
  static bool get hasProfile => _box.get('profile') != null;

  static Map<String, dynamic> get profile =>
      Map<String, dynamic>.from(_box.get('profile') ?? {});

  static Future<void> saveProfile(Map<String, dynamic> p) =>
      _box.put('profile', p);

  static Future<void> updateProfile(Map<String, dynamic> changes) =>
      _box.put('profile', {...profile, ...changes});

  // ---------- الإعدادات ----------
  static Map<String, dynamic> get _settings =>
      Map<String, dynamic>.from(_box.get('settings') ?? {});

  static T setting<T>(String key, T fallback) =>
      (_settings[key] as T?) ?? fallback;

  static Future<void> setSetting(String key, dynamic value) async {
    final s = _settings;
    s[key] = value;
    await _box.put('settings', s);
  }

  static Map<String, dynamic> get allSettings => _settings;

  static ThemeMode get themeMode {
    switch (setting<String>('theme', 'dark')) {
      case 'light':
        return ThemeMode.light;
      case 'system':
        return ThemeMode.system;
      default:
        return ThemeMode.dark;
    }
  }

  static Color get accent => Color(setting<int>('accent', 0xFF6C5CE7));

  // ---------- تواريخ الفصول الدراسية ----------
  static Map<String, dynamic> get termRanges =>
      Map<String, dynamic>.from((_settings['termRanges'] as Map?) ?? {});

  static Map<String, dynamic>? termRange(String level, String term) {
    final r = termRanges['$level|$term'];
    return r == null ? null : Map<String, dynamic>.from(r as Map);
  }

  static Future<void> setTermRange(String level, String term, String start, String end) async {
    final all = termRanges;
    all['$level|$term'] = {'start': start, 'end': end};
    await setSetting('termRanges', all);
  }

  // ---------- القوائم (مواد، محاضرات، حضور) ----------
  static List<Map<String, dynamic>> all(String key) {
    final raw = _box.get(key);
    if (raw == null) return [];
    return (raw['items'] as List)
        .map((e) => Map<String, dynamic>.from(e as Map))
        .toList();
  }

  static Future<void> upsert(String key, Map<String, dynamic> item) async {
    final items = all(key);
    final i = items.indexWhere((e) => e['id'] == item['id']);
    if (i >= 0) {
      items[i] = item;
    } else {
      items.add(item);
    }
    await _box.put(key, {'items': items});
  }

  static Future<void> remove(String key, String id) async {
    final items = all(key)..removeWhere((e) => e['id'] == id);
    await _box.put(key, {'items': items});
  }

  static Map<String, dynamic>? subject(String id) {
    for (final s in all('subjects')) {
      if (s['id'] == id) return s;
    }
    return null;
  }

  static String subjectName(String id) =>
      (subject(id)?['name'] as String?) ?? 'مادة محذوفة';

  // ---------- الحضور ----------
  static String? attendanceStatus(String lectureId, String date) {
    for (final a in all('attendance')) {
      if (a['lectureId'] == lectureId && a['date'] == date) {
        return a['status'] as String;
      }
    }
    return null;
  }

  static Future<void> setAttendance(
      String lectureId, String date, String? status) {
    final id = '$lectureId|$date';
    if (status == null) return remove('attendance', id);
    return upsert('attendance', {
      'id': id,
      'lectureId': lectureId,
      'date': date,
      'status': status,
    });
  }

  /// نسبة الحضور لمادة (الملغاة لا تُحتسب). تعيد null إذا لا توجد سجلات.
  static int? attendanceRate(String subjectId) {
    final lectureIds = all('lectures')
        .where((l) => l['subjectId'] == subjectId)
        .map((l) => l['id'])
        .toSet();
    var total = 0;
    var attended = 0;
    for (final a in all('attendance')) {
      if (!lectureIds.contains(a['lectureId']) || a['status'] == 'ملغاة') {
        continue;
      }
      total++;
      if (a['status'] == 'حاضر' || a['status'] == 'متأخر') attended++;
    }
    return total == 0 ? null : (attended * 100 ~/ total);
  }

  // ---------- الحذف المتسلسل ----------
  static Future<void> deleteLecture(String id) async {
    for (final a in all('attendance').where((a) => a['lectureId'] == id).toList()) {
      await remove('attendance', a['id'] as String);
    }
    await remove('lectures', id);
  }

  static Future<void> deleteSubject(String id) async {
    for (final l in all('lectures').where((l) => l['subjectId'] == id).toList()) {
      await deleteLecture(l['id'] as String);
    }
    await remove('subjects', id);
    await remove('notes', id);
  }

  // ---------- النسخ الاحتياطي ----------
  static const backupKeys = ['subjects', 'lectures', 'attendance', 'tasks', 'exams', 'notes'];

  static Map<String, dynamic> exportData() => {
        'version': 1,
        'profile': profile,
        'settings': _settings,
        for (final key in backupKeys) key: all(key),
      };

  static Future<void> restore(Map<String, dynamic> data) async {
    await _box.clear();
    final profileData = data['profile'];
    if (profileData is Map) {
      await _box.put('profile', Map<String, dynamic>.from(profileData));
    }
    final settingsData = data['settings'];
    if (settingsData is Map) {
      await _box.put('settings', Map<String, dynamic>.from(settingsData));
    }
    for (final key in backupKeys) {
      final items = data[key];
      await _box.put(key, {'items': items is List ? items : []});
    }
  }

  static Future<void> wipe() => _box.clear();
}

DateTime dateForDayInThisWeek(int dayIndex) {
  final now = DateTime.now();
  final today = DateTime(now.year, now.month, now.day);
  return today.add(Duration(days: dayIndex - dayIndexOf(today)));

  AppTheme._();

  static ThemeData light(Color accent) => _build(Brightness.light, accent);
  static ThemeData dark(Color accent) => _build(Brightness.dark, accent);

  static ThemeData _build(Brightness brightness, Color accent) {
    final isDark = brightness == Brightness.dark;
    final scheme = ColorScheme.fromSeed(seedColor: accent, brightness: brightness);
    return ThemeData(
      useMaterial3: true,
      colorScheme: scheme,
      scaffoldBackgroundColor: isDark ? const Color(0xFF11121A) : const Color(0xFFF3F4FA),
      appBarTheme: AppBarTheme(
        elevation: 0,
        scrolledUnderElevation: 0,
        backgroundColor: Colors.transparent,
        surfaceTintColor: Colors.transparent,
        titleTextStyle: TextStyle(
          fontSize: 24,
          fontWeight: FontWeight.w900,
          color: scheme.onSurface,
        ),
      ),
      inputDecorationTheme: InputDecorationTheme(
        filled: true,
        fillColor: isDark ? const Color(0xFF232637) : const Color(0xFFF7F8FC),

Future<String?> pickProfilePhoto() async {
  final file = await ImagePicker().pickImage(
    source: ImageSource.gallery,
    maxWidth: 256,
    maxHeight: 256,
    imageQuality: 70,
  );
  if (file == null) return null;
  return base64Encode(await file.readAsBytes());

ImageProvider? photoProvider(String? encoded) {
  if (encoded == null || encoded.isEmpty) return null;
  return MemoryImage(base64Decode(encoded));
}

import 'dart:async';
import 'package:flutter/material.dart';

/// مؤقت المذاكرة: 25 دقيقة تركيز ثم 5 دقائق استراحة.
class PomodoroScreen extends StatefulWidget {
  const PomodoroScreen({super.key});

  @override
  State<PomodoroScreen> createState() => _PomodoroScreenState();
}

class _PomodoroScreenState extends State<PomodoroScreen> {
  static const _workSeconds = 25 * 60;
  static const _breakSeconds = 5 * 60;

  Timer? _timer;
  bool _isWork = true;
  int _left = _workSeconds;
  int _sessions = 0;

  bool get _running => _timer != null;

  void _start() {
    _timer = Timer.periodic(const Duration(seconds: 1), (_) {
      if (_left <= 1) {
        _timer?.cancel();
        setState(() {
          _timer = null;
          if (_isWork) _sessions++;
          _isWork = !_isWork;
          _left = _isWork ? _workSeconds : _breakSeconds;
        });
      } else {
        setState(() {
          _left--;
        });
      }
    });
    setState(() {});
  }

  void _pause() {
    _timer?.cancel();
    setState(() {
      _timer = null;
    });
  }

  void _reset() {
    _timer?.cancel();
    setState(() {
      _timer = null;
      _left = _isWork ? _workSeconds : _breakSeconds;
    });
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();

import 'package:flutter/foundation.dart';
import 'package:flutter_local_notifications/flutter_local_notifications.dart';

/// التنبيهات المحلية: تعمل بدون إنترنت، وتستخدم صوت التنبيه الافتراضي (alarm)
/// الذي يرفعه المستخدم في مجلد sounds بالمستودع.
class Notifications {
  Notifications._();

  static final _plugin = FlutterLocalNotificationsPlugin();
  static const _channelId = 'unimate_alerts';
  static const _channelName = 'تنبيهات رفيق الطالب الجامعي';
  static const _soundName = 'alarm';
  static bool _ready = false;
  static ValueListenable<dynamic>? _watcher;

  static Future<void> init() async {
    tzdata.initializeTimeZones();
    tz.setLocalLocation(tz.getLocation('Asia/Aden'));

    const settings = InitializationSettings(
      android: AndroidInitializationSettings('@mipmap/ic_launcher'),
    );
    await _plugin.initialize(settings);

    final android = _plugin.resolvePlatformSpecificImplementation<
        AndroidFlutterLocalNotificationsPlugin>();
    final channel = AndroidNotificationChannel(
      _channelId,
      _channelName,
      importance: Importance.max,
      sound: const RawResourceAndroidNotificationSound(_soundName),
      playSound: true,
    );
    await android?.createNotificationChannel(channel);
    await android?.requestNotificationsPermission();
    await android?.requestExactAlarmsPermission();
    _ready = true;
  }

  /// يعيد جدولة كل التنبيهات من البيانات الحالية. يُستدعى عند كل تغيير.
  static void watchStore() {
    _watcher = Store.listenable;
    _watcher!.addListener(rescheduleAll);
  }

  static Future<void> rescheduleAll() async {
    if (!_ready) return;
    await _plugin.cancelAll();
    if (Store.setting<bool>('alerts', true) == false) return;

    final now = tz.TZDateTime.now(tz.local);
    final details = NotificationDetails(
      android: AndroidNotificationDetails(
        _channelId,
        _channelName,
        importance: Importance.max,
        priority: Priority.high,
        sound: const RawResourceAndroidNotificationSound(_soundName),
        playSound: true,
      ),
    );

    // المحاضرات: تتكرر أسبوعياً، قبل بدايتها بالمدة التي يختارها المستخدم.
    final lead = Store.setting<int>('lectureLead', 10);
    for (final l in Store.all('lectures')) {
      final minutes = toMin(l['start'] as String) - lead;
      if (minutes < 0) continue;
      final when = _nextWeekly(l['day'] as int, minutes);
      await _plugin.zonedSchedule(
        _idOf('lec|${l['id']}'),
        'محاضرة بعد $lead دقيقة',
        '${Store.subjectName(l['subjectId'] as String)} - ${l['start']}',
        when,
        details,
        androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
        uiLocalNotificationDateInterpretation:
            UILocalNotificationDateInterpretation.absoluteTime,
        matchDateTimeComponents: DateTimeComponents.dayOfWeekAndTime,
      );
    }

    // الاختبارات: قبلها بساعة، ويوم قبلها الساعة 9 صباحاً.
    for (final e in Store.all('exams')) {
      final date = DateTime.parse(e['date'] as String);
      final t = toMin(e['time'] as String);
      final examAt = tz.TZDateTime(
        tz.local,
        date.year,
        date.month,
        date.day,
        t ~/ 60,
        t % 60,
      );
      final name = Store.subjectName(e['subjectId'] as String);
      await _once(
        _idOf('exam1|${e['id']}'),
        'اختبار بعد ساعة',
        '$name - ${e['type']}',
        examAt.subtract(const Duration(hours: 1)),
        details,
        now,
      );
      await _once(
        _idOf('exam2|${e['id']}'),
        'اختبار غداً',
        '$name - ${e['type']} الساعة ${e['time']}',
        tz.TZDateTime(tz.local, date.year, date.month, date.day - 1, 9),
        details,
        now,
      );
    }

    // المهام غير المكتملة: في الساعة التي يختارها المستخدم من يوم موعدها.
    for (final t in Store.all('tasks')) {
      if (t['status'] == 'مكتملة') continue;
      final due = DateTime.parse(t['due'] as String);
      await _once(
        _idOf('task|${t['id']}'),
        'مهمة اليوم',
        t['title'] as String,
        tz.TZDateTime(tz.local, due.year, due.month, due.day, Store.setting<int>('taskHour', 8)),
        details,
        now,
      );
    }
  }

  static Future<void> _once(
    int id,
    String title,
    String body,
    tz.TZDateTime when,
    NotificationDetails details,
    tz.TZDateTime now,
  ) async {
    if (!when.isAfter(now)) return;
    await _plugin.zonedSchedule(
      id,
      title,
      body,
      when,
      details,
      androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
      uiLocalNotificationDateInterpretation:
          UILocalNotificationDateInterpretation.absoluteTime,
    );
  }

  /// أقرب موعد أسبوعي لليوم المحدد (السبت = 0) والدقائق المحددة.
  static tz.TZDateTime _nextWeekly(int dayIdx, int minutes) {
    final weekday = (dayIdx + 5) % 7 + 1; // تحويل إلى ترقيم Dart (الإثنين = 1)
    final now = tz.TZDateTime.now(tz.local);
    var d = tz.TZDateTime(
      tz.local,
      now.year,
      now.month,
      now.day,
      minutes ~/ 60,
      minutes % 60,
    );
    while (d.weekday != weekday || !d.isAfter(now)) {
      d = d.add(const Duration(days: 1));
    }
    return d;
  }

  static int _idOf(String key) => key.hashCode & 0x3fffffff;
